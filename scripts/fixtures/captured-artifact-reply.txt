  <script>
    // Theme toggle
    const html = document.documentElement;
    const btn = document.getElementById('themeToggle');
    const sun = document.getElementById('iconSun');
    const moon = document.getElementById('iconMoon');

    const stored = localStorage.getItem('theme');
    if (stored === 'dark' || (!stored && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
      html.classList.add('dark');
      sun.classList.remove('hidden');
      moon.classList.add('hidden');
    }

    btn.addEventListener('click', () => {
      const isDark = html.classList.contains('dark');
      if (isDark) {
        html.classList.remove('dark');
        sun.classList.add('hidden');
        moon.classList.remove('hidden');
        localStorage.setItem('theme', 'light');
      } else {
        html.classList.add('dark');
        sun.classList.remove('hidden');
        moon.classList.add('hidden');
        localStorage.setItem('theme', 'dark');
      }
    });

    // Billing toggle
    const toggle = document.getElementById('billingToggle');
    const cardM = document.getElementById('cardMonthly');
    const cardA = document.getElementById('cardAnnual');

    toggle.addEventListener('click', () => {
      toggle.classList.toggle('active');
      if (toggle.classList.contains('active')) {
        cardM.classList.add('hidden');
        cardA.classList.remove('hidden');
      } else {
        cardM.classList.remove('hidden');
        cardA.classList.add('hidden');
      }
    });
  </script>
